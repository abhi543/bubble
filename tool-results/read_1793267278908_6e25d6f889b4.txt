   543→  /* One little note with a soft envelope */
   544→  private tone(o: {
   545→    freq: number
   546→    time: number
   547→    dur: number
   548→    type?: OscillatorType
   549→    vol?: number
   550→    glide?: number
   551→    dest?: GainNode | null
   552→  }) {
   553→    if (!this.ctx || !this.master) return
   554→    const { freq, time, dur, type = 'sine', vol = 0.2, glide, dest = this.master } = o
   555→    const t = Math.max(time, this.ctx.currentTime)
   556→    const osc = this.ctx.createOscillator()
   557→    const gain = this.ctx.createGain()
   558→    osc.type = type
   559→    osc.frequency.setValueAtTime(freq, t)
   560→    if (glide) osc.frequency.exponentialRampToValueAtTime(glide, t + dur)
   561→    gain.gain.setValueAtTime(0.0001, t)
   562→    gain.gain.exponentialRampToValueAtTime(vol, t + 0.015)
   563→    gain.gain.exponentialRampToValueAtTime(0.0001, t + dur)
   564→    osc.connect(gain)
   565→    gain.connect(dest)
   566→    osc.start(t)
   567→    osc.stop(t + dur + 0.05)
   568→  }

   708→     32 steps of 8th-notes at 120 BPM. Chords: C → Am → F → G.          */
   709→  startMusic(fast = false) {
   710→    this.init()
   711→    if (!this.ctx || this.musicTimer) return
   712→    this.stepDur = fast ? 0.227 : 0.25 // 132 BPM vs 120 BPM
   713→    this.step = 0
   714→    this.nextTime = this.ctx.currentTime + 0.1
   715→    this.musicTimer = setInterval(() => this.scheduleMusic(), 100)
   716→  }

   718→  stopMusic() {
   719→    if (this.musicTimer) {
   720→      clearInterval(this.musicTimer)
   721→      this.musicTimer = null
   722→    }
   723→  }

   725→  private scheduleMusic() {
   726→    if (!this.ctx) return
   727→    const STEP = this.stepDur // one 8th-note at the current tempo
   728→    while (this.nextTime < this.ctx.currentTime + 0.4) {
   729→      this.playMusicStep(this.step, this.nextTime)
   730→      this.step = (this.step + 1) % 32
   731→      this.nextTime += STEP
   732→    }
   733→  }

   735→  private playMusicStep(s: number, t: number) {
   736→    if (!this.musicGain) return
   737→    // melody (midi notes, 0 = rest) — 4 bars of pure pentatonic happiness
   738→    const MEL = [72, 0, 76, 0, 79, 0, 76, 0, 81, 0, 79, 0, 76, 0, 74, 0, 72, 0, 76, 0, 79, 0, 81, 0, 79, 0, 76, 0, 74, 0, 72, 0]
   739→    // bass roots: C  A  F  G (one per bar)
   740→    const BASS = [48, 45, 41, 43]
   741→    if (MEL[s]) this.tone({ freq: this.f(MEL[s]), time: t, dur: 0.22, type: 'triangle', vol: 0.15, dest: this.musicGain })
   742→    if (s % 8 === 0) this.tone({ freq: this.f(BASS[(s / 8) | 0]), time: t, dur: 0.5, type: 'sine', vol: 0.2, dest: this.musicGain })
   743→    if (s % 8 === 4) this.tone({ freq: this.f(BASS[(s / 8) | 0]) + 12, time: t, dur: 0.3, type: 'sine', vol: 0.1, dest: this.musicGain })
   744→    if (s % 2 === 1) this.tone({ freq: 2100, time: t, dur: 0.03, vol: 0.025, dest: this.musicGain }) // tiny bounce tick
   745→  }
   746→}

   757→function roundRectPath(ctx: CanvasRenderingContext2D, x: number, y: number, w: number, h: number, r: number) {
   758→  ctx.beginPath()
   759→  ctx.moveTo(x + r, y)
   760→  ctx.arcTo(x + w, y, x + w, y + h, r)
   761→  ctx.arcTo(x + w, y + h, x, y + h, r)

   767→/* A five-pointed (or any-pointed!) star path */
   768→function starPath(
   769→  ctx: CanvasRenderingContext2D,
   770→  x: number,
   771→  y: number,
   772→  outer: number,
   773→  inner: number,
   774→  points: number,
   775→  rot: number
   776→) {
   777→  ctx.beginPath()
   778→  for (let i = 0; i < points * 2; i++) {
   779→    const rr = i % 2 === 0 ? outer : inner
   780→    const a = rot + (i * Math.PI) / points
   781→    const px = x + Math.cos(a) * rr
   782→    const py = y + Math.sin(a) * rr
   783→    if (i === 0) ctx.moveTo(px, py)
   784→    else ctx.lineTo(px, py)
   785→  }
   786→  ctx.closePath()
   787→}

   789→/* 🥰 THE PAL'S FACE — three moods, used by all four characters:
   790→     smile     = happy default (with cute blinks)
   791→     surprised = "uh-oh!" round eyes + little O mouth (after losing a heart)
   792→     cheer     = squeezed-shut happy eyes + big open grin (level complete!)
   793→   dx / dy move the face inside the character (like Rocky's window). */
   794→function palFace(ctx: CanvasRenderingContext2D, r: number, expr: FaceExpr, blink: boolean, dx = 0, dy = 0) {
   795→  ctx.save()
   796→  ctx.translate(dx, dy)
   797→  const ex = r * 0.36 // eye spacing
   798→  const ey = -r * 0.12 // eye height
   799→  const er = r * 0.22 // eye radius
   800→  if (expr === 'cheer') {
   801→    // ^ ^  two happy squeezed-shut eyes
   802→    ctx.strokeStyle = COLORS.face
   803→    ctx.lineWidth = 2.5
   804→    ctx.lineCap = 'round'
   805→    for (const side of [-1, 1]) {
   806→      ctx.beginPath()
   807→      ctx.arc(side * ex, ey, er, Math.PI * 1.1, Math.PI * 1.9)
   808→      ctx.stroke()
   809→    }
   810→  } else {
   811→    // big white eyes — extra wide when surprised!
   812→    const big = expr === 'surprised' ? 1.3 : 1
   813→    ctx.fillStyle = '#ffffff'
   814→    circle(ctx, -ex, ey, er * big)
   815→    circle(ctx, ex, ey, er * big)
   816→    // pupils — tiny and high when surprised, low and calm when smiling
   817→    ctx.fillStyle = COLORS.face
   818→    const pr = expr === 'surprised' ? er * 0.3 : er * 0.45
   819→    const py = expr === 'surprised' ? ey - er * 0.15 : ey + er * 0.25
   820→    circle(ctx, -ex, py, pr)
   821→    circle(ctx, ex, py, pr)
   822→  }
   823→  // the mouth
   824→  if (expr === 'surprised') {
   825→    // 😮 a little round "O" mouth
   826→    ctx.fillStyle = COLORS.face
   827→    circle(ctx, 0, r * 0.24, r * 0.16)
   828→  } else if (expr === 'cheer') {
   829→    // 🥳 a big open grin
   830→    ctx.fillStyle = COLORS.face
   831→    ctx.beginPath()
   832→    ctx.arc(0, r * 0.06, r * 0.3, Math.PI * 0.12, Math.PI * 0.88)
   833→    ctx.closePath()
   834→    ctx.fill()
   835→  } else {
   836→    // 🙂 the classic smile
   837→    ctx.strokeStyle = COLORS.face
   838→    ctx.lineWidth = 2.5
   839→    ctx.lineCap = 'round'
   840→    ctx.beginPath()
   841→    ctx.arc(0, r * 0.12, r * 0.3, Math.PI * 0.2, Math.PI * 0.8)
   842→    ctx.stroke()
   843→  }
   844→  // rosy cheeks — extra rosy when cheering!
   845→  ctx.fillStyle = expr === 'cheer' ? 'rgba(255,110,150,0.5)' : 'rgba(255,110,150,0.35)'
   846→  circle(ctx, -r * 0.6, r * 0.1, r * 0.14)
   847→  circle(ctx, r * 0.6, r * 0.1, r * 0.14)
   848→  ctx.restore()
   849→}

  1593→function updateParticles(g: GameData, dt: number) {
  1594→  for (let i = g.parts.length - 1; i >= 0; i--) {
  1595→    const p = g.parts[i]
  1596→    p.life -= dt
  1597→    if (p.life <= 0) {
  1598→      g.parts.splice(i, 1)
  1599→      continue
  1600→    }
  1601→    p.x += p.vx * dt
  1602→    p.y += p.vy * dt
  1603→    p.rot += p.vr * dt
  1604→    if (p.kind === 'confetti') {
  1605→      p.vy += 420 * dt // confetti tumbles down
  1606→      p.vx *= 1 - 1.2 * dt
  1607→    } else if (p.kind === 'sparkle') {
  1608→      p.vy += 140 * dt
  1609→    } else if (p.kind === 'ring') {
  1610→      p.size += 190 * dt // ring grows outward
  1611→    }
  1612→  }
  1613→}

  1615→function drawParticles(ctx: CanvasRenderingContext2D, g: GameData) {
  1616→  for (const p of g.parts) {
  1617→    const a = clamp(p.life / p.maxLife, 0, 1)
  1618→    ctx.globalAlpha = a
  1619→    if (p.kind === 'confetti') {
  1620→      ctx.save()
  1621→      ctx.translate(p.x, p.y)
  1622→      ctx.rotate(p.rot)
  1623→      ctx.fillStyle = p.color
  1624→      ctx.fillRect(-p.size / 2, -p.size / 2, p.size, p.size * 0.62)
  1625→      ctx.restore()
  1626→    } else if (p.kind === 'sparkle') {
  1627→      ctx.fillStyle = p.color
  1628→      starPath(ctx, p.x, p.y, p.size + 2, (p.size + 2) * 0.4, 4, p.rot)
  1629→      ctx.fill()
  1630→    } else if (p.kind === 'ring') {
  1631→      ctx.strokeStyle = p.color
  1632→      ctx.lineWidth = 3.5 * a
  1633→      ctx.beginPath()
  1634→      ctx.arc(p.x, p.y, p.size, 0, Math.PI * 2)
  1635→      ctx.stroke()
  1636→    } else if (p.kind === 'text' && p.text) {
  1637→      ctx.font = `bold ${p.size || 22}px ${FONT}`
  1638→      ctx.textAlign = 'center'
  1639→      ctx.textBaseline = 'middle'
  1640→      ctx.lineWidth = 4
  1641→      ctx.strokeStyle = 'rgba(255,255,255,0.9)'
  1642→      ctx.strokeText(p.text, p.x, p.y)
  1643→      ctx.fillStyle = p.color
  1644→      ctx.fillText(p.text, p.x, p.y)
  1645→    }
  1646→    ctx.globalAlpha = 1
  1647→  }
  1648→}

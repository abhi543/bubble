   253→class GameAudio {
   254→  private ctx: AudioContext | null = null
   255→  private master: GainNode | null = null
   256→  private musicGain: GainNode | null = null
   257→  private musicTimer: ReturnType<typeof setInterval> | null = null
   258→  private step = 0
   259→  private nextTime = 0
   260→  private stepDur = 0.25 // one 8th-note; grown-ups mode gets a pepzier tempo
   261→  muted = false
   262→
   263→  /* Create the audio context lazily (browsers require a user gesture first) */
   264→  init() {
   265→    if (typeof window === 'undefined') return
   266→    try {
   267→      if (!this.ctx) {
   268→        const AC =
   269→          window.AudioContext ||
   270→          (window as unknown as { webkitAudioContext: typeof AudioContext }).webkitAudioContext
   271→        this.ctx = new AC()
   272→        this.master = this.ctx.createGain()
   273→        this.master.gain.value = this.muted ? 0 : 0.9
   274→        this.master.connect(this.ctx.destination)
   275→        this.musicGain = this.ctx.createGain()
   276→        this.musicGain.gain.value = 0.5 // music a bit quieter than effects
   277→        this.musicGain.connect(this.master)
   278→      }
   279→      if (this.ctx.state === 'suspended') void this.ctx.resume()
   280→    } catch {
   281→      /* no audio available — the game still works silently */
   282→    }
   283→  }
   284→
   285→  setMuted(m: boolean) {
   286→    this.muted = m
   287→    if (this.master && this.ctx) this.master.gain.setTargetAtTime(m ? 0 : 0.9, this.ctx.currentTime, 0.02)
   288→  }
   289→
   290→  private f(midi: number) {
   291→    return 440 * Math.pow(2, (midi - 69) / 12)
   292→  }
   293→
   294→  /* One little note with a soft envelope */
   295→  private tone(o: {
   296→    freq: number
   297→    time: number
   298→    dur: number
   299→    type?: OscillatorType
   300→    vol?: number
   301→    glide?: number
   302→    dest?: GainNode | null
   303→  }) {
   304→    if (!this.ctx || !this.master) return
   305→    const { freq, time, dur, type = 'sine', vol = 0.2, glide, dest = this.master } = o
   306→    const t = Math.max(time, this.ctx.currentTime)
   307→    const osc = this.ctx.createOscillator()
   308→    const gain = this.ctx.createGain()
   309→    osc.type = type
   310→    osc.frequency.setValueAtTime(freq, t)
   311→    if (glide) osc.frequency.exponentialRampToValueAtTime(glide, t + dur)
   312→    gain.gain.setValueAtTime(0.0001, t)
   313→    gain.gain.exponentialRampToValueAtTime(vol, t + 0.015)
   314→    gain.gain.exponentialRampToValueAtTime(0.0001, t + dur)
   315→    osc.connect(gain)
   316→    gain.connect(dest)
   317→    osc.start(t)
   318→    osc.stop(t + dur + 0.05)
   319→  }
   320→
   321→  /* ── SOUND EFFECTS ────────────────────────────────────────────────── */
   322→  click() {
   323→    this.init()
   324→    if (!this.ctx) return
   325→    this.tone({ freq: 750, time: this.ctx.currentTime, dur: 0.07, vol: 0.12, glide: 980 })
   326→  }
   327→
   328→  /* soft bubbly "pew" when shooting */
   329→  shoot() {
   330→    this.init()
   331→    if (!this.ctx) return
   332→    this.tone({ freq: 340, time: this.ctx.currentTime, dur: 0.11, vol: 0.14, glide: 640 })
   333→  }
   334→
   335→  /* happy musical "pop!" — each pop chimes a different happy note */
   336→  pop() {
   337→    this.init()
   338→    if (!this.ctx) return
   339→    const t = this.ctx.currentTime
   340→    const base = pick([523.25, 587.33, 659.25, 783.99, 880])
   341→    this.tone({ freq: base, time: t, dur: 0.18, type: 'triangle', vol: 0.22 })
   342→    this.tone({ freq: base * 2, time: t + 0.04, dur: 0.12, vol: 0.09 })
   343→  }
   344→
   345→  /* gentle "oops" when a monster slips past — friendly, never scary */
   346→  oops() {
   347→    this.init()
   348→    if (!this.ctx) return
   349→    const t = this.ctx.currentTime
   350→    this.tone({ freq: 392, time: t, dur: 0.15, type: 'triangle', vol: 0.18 })
   351→    this.tone({ freq: 311, time: t + 0.16, dur: 0.28, type: 'triangle', vol: 0.18 })
   352→  }
   353→
   354→  /* little fanfare when a level is complete */
   355→  levelUp() {
   356→    this.init()
   357→    if (!this.ctx) return
   358→    const t = this.ctx.currentTime
   359→    const notes = [72, 76, 79, 84] // C5 E5 G5 C6
   360→    notes.forEach((n, i) =>
   361→      this.tone({ freq: this.f(n), time: t + i * 0.1, dur: i === 3 ? 0.45 : 0.18, type: 'triangle', vol: 0.2 })
   362→    )
   363→  }
   364→
   365→  /* BIG happy fanfare for finishing all three levels */
   366→  win() {
   367→    this.init()
   368→    if (!this.ctx) return
   369→    const t = this.ctx.currentTime
   370→    const seq: Array<[number, number, number]> = [
   371→      [72, 0, 0.14],
   372→      [76, 0.12, 0.14],
   373→      [79, 0.24, 0.14],
   374→      [84, 0.36, 0.3],
   375→      [79, 0.6, 0.12],
   376→      [84, 0.72, 0.14],
   377→      [88, 0.9, 0.6],
   378→    ]
   379→    seq.forEach(([n, dt, dur]) =>
   380→      this.tone({ freq: this.f(n), time: t + dt, dur, type: 'triangle', vol: 0.2 })
   381→    )
   382→  }
   383→
   384→  /* soft, encouraging jingle for "Great try!" — no sad trombones here */
   385→  gameOverJingle() {
   386→    this.init()
   387→    if (!this.ctx) return
   388→    const t = this.ctx.currentTime
   389→    this.tone({ freq: this.f(64), time: t, dur: 0.22, type: 'triangle', vol: 0.16 })
   390→    this.tone({ freq: this.f(60), time: t + 0.2, dur: 0.22, type: 'triangle', vol: 0.16 })
   391→    this.tone({ freq: this.f(57), time: t + 0.42, dur: 0.5, type: 'triangle', vol: 0.16 })
   392→  }
   393→
   394→  /* ── MUSIC LOOP 🎵 — a bouncy 8-bar tune that repeats forever ────────
   395→     32 steps of 8th-notes at 120 BPM. Chords: C → Am → F → G.          */
   396→  startMusic(fast = false) {
   397→    this.init()
   398→    if (!this.ctx || this.musicTimer) return
   399→    this.stepDur = fast ? 0.227 : 0.25 // 132 BPM vs 120 BPM
   400→    this.step = 0
   401→    this.nextTime = this.ctx.currentTime + 0.1
   402→    this.musicTimer = setInterval(() => this.scheduleMusic(), 100)
   403→  }
   404→
   405→  stopMusic() {
   406→    if (this.musicTimer) {
   407→      clearInterval(this.musicTimer)
   408→      this.musicTimer = null
   409→    }
   410→  }
   411→
   412→  private scheduleMusic() {
   413→    if (!this.ctx) return
   414→    const STEP = this.stepDur // one 8th-note at the current tempo
   415→    while (this.nextTime < this.ctx.currentTime + 0.4) {
   416→      this.playMusicStep(this.step, this.nextTime)
   417→      this.step = (this.step + 1) % 32
   418→      this.nextTime += STEP
   419→    }
   420→  }
   421→
   422→  private playMusicStep(s: number, t: number) {
   423→    if (!this.musicGain) return
   424→    // melody (midi notes, 0 = rest) — 4 bars of pure pentatonic happiness
   425→    const MEL = [72, 0, 76, 0, 79, 0, 76, 0, 81, 0, 79, 0, 76, 0, 74, 0, 72, 0, 76, 0, 79, 0, 81, 0, 79, 0, 76, 0, 74, 0, 72, 0]
   426→    // bass roots: C  A  F  G (one per bar)
   427→    const BASS = [48, 45, 41, 43]
   428→    if (MEL[s]) this.tone({ freq: this.f(MEL[s]), time: t, dur: 0.22, type: 'triangle', vol: 0.15, dest: this.musicGain })
   429→    if (s % 8 === 0) this.tone({ freq: this.f(BASS[(s / 8) | 0]), time: t, dur: 0.5, type: 'sine', vol: 0.2, dest: this.musicGain })
   430→    if (s % 8 === 4) this.tone({ freq: this.f(BASS[(s / 8) | 0]) + 12, time: t, dur: 0.3, type: 'sine', vol: 0.1, dest: this.musicGain })
   431→    if (s % 2 === 1) this.tone({ freq: 2100, time: t, dur: 0.03, vol: 0.025, dest: this.musicGain }) // tiny bounce tick
   432→  }
   433→}

   438→function circle(ctx: CanvasRenderingContext2D, x: number, y: number, r: number) {
   439→  ctx.beginPath()
   440→  ctx.arc(x, y, r, 0, Math.PI * 2)
   441→  ctx.fill()
   442→}

   444→function roundRectPath(ctx: CanvasRenderingContext2D, x: number, y: number, w: number, h: number, r: number) {
   445→  ctx.beginPath()
   446→  ctx.moveTo(x + r, y)
   447→  ctx.arcTo(x + w, y, x + w, y + h, r)
   448→  ctx.arcTo(x + w, y + h, x, y + h, r)
   449→  ctx.arcTo(x, y + h, x, y, r)
   450→  ctx.arcTo(x, y, x + w, y, r)
   451→  ctx.closePath()
   452→}

   454→/* A five-pointed (or any-pointed!) star path */
   455→function starPath(
   456→  ctx: CanvasRenderingContext2D,
   457→  x: number,
   458→  y: number,
   459→  outer: number,
   460→  inner: number,
   461→  points: number,
   462→  rot: number
   463→) {
   464→  ctx.beginPath()
   465→  for (let i = 0; i < points * 2; i++) {
   466→    const rr = i % 2 === 0 ? outer : inner
   467→    const a = rot + (i * Math.PI) / points
   468→    const px = x + Math.cos(a) * rr
   469→    const py = y + Math.sin(a) * rr
   470→    if (i === 0) ctx.moveTo(px, py)
   471→    else ctx.lineTo(px, py)
   472→  }
   473→  ctx.closePath()
   474→}

   476→/* Every character shares this happy face 🙂 */
   477→function happyFace(ctx: CanvasRenderingContext2D, r: number, blinking: boolean, scale = 1, dy = 0) {
   478→  const ex = r * 0.34 * scale
   479→  const ey = -r * 0.1 + dy
   480→  const er = r * 0.2 * scale
   481→  if (blinking) {
   482→    // eyes closed = two happy little arcs
   483→    ctx.strokeStyle = COLORS.face
   484→    ctx.lineWidth = 2.5
   485→    ctx.lineCap = 'round'
   486→    ctx.beginPath()
   487→    ctx.arc(-ex, ey, er * 0.9, Math.PI * 1.1, Math.PI * 1.9)
   488→    ctx.stroke()
   489→    ctx.beginPath()
   490→    ctx.arc(ex, ey, er * 0.9, Math.PI * 1.1, Math.PI * 1.9)
   491→    ctx.stroke()
   492→  } else {
   493→    ctx.fillStyle = '#ffffff'
   494→    circle(ctx, -ex, ey, er)
   495→    circle(ctx, ex, ey, er)
   496→    ctx.fillStyle = COLORS.face
   497→    circle(ctx, -ex, ey + er * 0.25, er * 0.45)
   498→    circle(ctx, ex, ey + er * 0.25, er * 0.45)
   499→  }
   500→  // the smile 🙂
   501→  ctx.strokeStyle = COLORS.face
   502→  ctx.lineWidth = 2.5

   670→function drawMonster(ctx: CanvasRenderingContext2D, m: Monster, time: number) {
   671→  const r = m.r
   672→  ctx.save()
   673→  ctx.translate(m.x, m.y)

   740→  ctx.restore()
   741→}

   823→function updateParticles(g: GameData, dt: number) {
   824→  for (let i = g.parts.length - 1; i >= 0; i--) {
   825→    const p = g.parts[i]
   826→    p.life -= dt
   827→    if (p.life <= 0) {
   828→      g.parts.splice(i, 1)
   829→      continue
   830→    }
   831→    p.x += p.vx * dt
   832→    p.y += p.vy * dt
   833→    p.rot += p.vr * dt
   834→    if (p.kind === 'confetti') {
   835→      p.vy += 420 * dt // confetti tumbles down
   836→      p.vx *= 1 - 1.2 * dt
   837→    } else if (p.kind === 'sparkle') {
   838→      p.vy += 140 * dt
   839→    } else if (p.kind === 'ring') {
   840→      p.size += 190 * dt // ring grows outward
   841→    }
   842→  }
   843→}
   844→
   845→function drawParticles(ctx: CanvasRenderingContext2D, g: GameData) {
   846→  for (const p of g.parts) {
   847→    const a = clamp(p.life / p.maxLife, 0, 1)
   848→    ctx.globalAlpha = a
   849→    if (p.kind === 'confetti') {
   850→      ctx.save()
   851→      ctx.translate(p.x, p.y)
   852→      ctx.rotate(p.rot)
   853→      ctx.fillStyle = p.color
   854→      ctx.fillRect(-p.size / 2, -p.size / 2, p.size, p.size * 0.62)
   855→      ctx.restore()
   856→    } else if (p.kind === 'sparkle') {
   857→      ctx.fillStyle = p.color
   858→      starPath(ctx, p.x, p.y, p.size + 2, (p.size + 2) * 0.4, 4, p.rot)
   859→      ctx.fill()
   860→    } else if (p.kind === 'ring') {
   861→      ctx.strokeStyle = p.color
   862→      ctx.lineWidth = 3.5 * a
   863→      ctx.beginPath()
   864→      ctx.arc(p.x, p.y, p.size, 0, Math.PI * 2)
   865→      ctx.stroke()
   866→    } else if (p.kind === 'text' && p.text) {
   867→      ctx.font = `bold 22px ${FONT}`
   868→      ctx.textAlign = 'center'
   869→      ctx.textBaseline = 'middle'
   870→      ctx.lineWidth = 4
   871→      ctx.strokeStyle = 'rgba(255,255,255,0.9)'
   872→      ctx.strokeText(p.text, p.x, p.y)
   873→      ctx.fillStyle = p.color
   874→      ctx.fillText(p.text, p.x, p.y)
   875→    }
   876→    ctx.globalAlpha = 1
   877→  }
   878→}

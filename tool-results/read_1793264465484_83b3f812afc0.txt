   344→/* ═══════════════════════════════════════════════════════════════════════
   345→   🔊 SOUND ENGINE — every beep is generated with the Web Audio API,
   346→   so there are no sound files to download. Tweak the numbers freely!
   347→   ═══════════════════════════════════════════════════════════════════════ */
   348→class GameAudio {
   349→  private ctx: AudioContext | null = null
   350→  private master: GainNode | null = null
   351→  private musicGain: GainNode | null = null
   352→  private musicTimer: ReturnType<typeof setInterval> | null = null
   353→  private step = 0
   354→  private nextTime = 0
   355→  private stepDur = 0.25 // one 8th-note; grown-ups mode gets a pepzier tempo
   356→  muted = false
   357→
   358→  /* Create the audio context lazily (browsers require a user gesture first) */
   359→  init() {
   360→    if (typeof window === 'undefined') return
   361→    try {
   362→      if (!this.ctx) {
   363→        const AC =
   364→          window.AudioContext ||
   365→          (window as unknown as { webkitAudioContext: typeof AudioContext }).webkitAudioContext
   366→        this.ctx = new AC()
   367→        this.master = this.ctx.createGain()
   368→        this.master.gain.value = this.muted ? 0 : 0.9
   369→        this.master.connect(this.ctx.destination)
   370→        this.musicGain = this.ctx.createGain()
   371→        this.musicGain.gain.value = 0.5 // music a bit quieter than effects
   372→        this.musicGain.connect(this.master)
   373→      }
   374→      if (this.ctx.state === 'suspended') void this.ctx.resume()
   375→    } catch {
   376→      /* no audio available — the game still works silently */
   377→    }
   378→  }
   379→
   380→  setMuted(m: boolean) {
   381→    this.muted = m
   382→    if (this.master && this.ctx) this.master.gain.setTargetAtTime(m ? 0 : 0.9, this.ctx.currentTime, 0.02)
   383→  }
   384→
   385→  private f(midi: number) {
   386→    return 440 * Math.pow(2, (midi - 69) / 12)
   387→  }
   388→
   389→  /* One little note with a soft envelope */
   390→  private tone(o: {
   391→    freq: number
   392→    time: number
   393→    dur: number
   394→    type?: OscillatorType
   395→    vol?: number
   396→    glide?: number
   397→    dest?: GainNode | null
   398→  }) {
   399→    if (!this.ctx || !this.master) return
   400→    const { freq, time, dur, type = 'sine', vol = 0.2, glide, dest = this.master } = o
   401→    const t = Math.max(time, this.ctx.currentTime)
   402→    const osc = this.ctx.createOscillator()
   403→    const gain = this.ctx.createGain()
   404→    osc.type = type
   405→    osc.frequency.setValueAtTime(freq, t)
   406→    if (glide) osc.frequency.exponentialRampToValueAtTime(glide, t + dur)
   407→    gain.gain.setValueAtTime(0.0001, t)
   408→    gain.gain.exponentialRampToValueAtTime(vol, t + 0.015)
   409→    gain.gain.exponentialRampToValueAtTime(0.0001, t + dur)
   410→    osc.connect(gain)
   411→    gain.connect(dest)
   412→    osc.start(t)
   413→    osc.stop(t + dur + 0.05)
   414→  }
   415→
   416→  /* ── SOUND EFFECTS ────────────────────────────────────────────────── */
   417→  click() {
   418→    this.init()
   419→    if (!this.ctx) return
   420→    this.tone({ freq: 750, time: this.ctx.currentTime, dur: 0.07, vol: 0.12, glide: 980 })
   421→  }
   422→
   423→  /* soft bubbly "pew" when shooting */
   424→  shoot() {
   425→    this.init()
   426→    if (!this.ctx) return
   427→    this.tone({ freq: 340, time: this.ctx.currentTime, dur: 0.11, vol: 0.14, glide: 640 })
   428→  }
   429→
   430→  /* happy musical "pop!" — each pop chimes a different happy note */
   431→  pop() {
   432→    this.init()
   433→    if (!this.ctx) return
   434→    const t = this.ctx.currentTime
   435→    const base = pick([523.25, 587.33, 659.25, 783.99, 880])
   436→    this.tone({ freq: base, time: t, dur: 0.18, type: 'triangle', vol: 0.22 })
   437→    this.tone({ freq: base * 2, time: t + 0.04, dur: 0.12, vol: 0.09 })
   438→  }
   439→
   440→  /* Every pal has their OWN happy pop! 🎵
   441→     rocky = bright ding · dolly = watery blub-blub
   442→     bunny = soft boing  · zippy = arcade bleep-bloop
   443→     Change the notes freely — sillier is better! */
   444→  popFor(char: CharacterId) {
   445→    this.init()
   446→    if (!this.ctx) return
   447→    const t = this.ctx.currentTime
   448→    if (char === 'rocky') {
   449→      // ⭐ a bright little ding-ding!
   450→      this.tone({ freq: 784, time: t, dur: 0.14, type: 'triangle', vol: 0.22 })
   451→      this.tone({ freq: 1568, time: t + 0.04, dur: 0.1, vol: 0.08 })
   452→    } else if (char === 'dolly') {
   453→      // 💧 a watery blub-blub (slides up like a droplet)
   454→      this.tone({ freq: 300, time: t, dur: 0.16, type: 'sine', vol: 0.22, glide: 620 })
   455→      this.tone({ freq: 520, time: t + 0.07, dur: 0.1, type: 'sine', vol: 0.1, glide: 740 })
   456→    } else if (char === 'bunny') {
   457→      // 💗 a soft springy boing
   458→      this.tone({ freq: 440, time: t, dur: 0.18, type: 'triangle', vol: 0.22, glide: 660 })
   459→    } else {
   460→      // 🌈 an arcade bleep-bloop
   461→      this.tone({ freq: 660, time: t, dur: 0.08, type: 'square', vol: 0.1 })
   462→      this.tone({ freq: 990, time: t + 0.06, dur: 0.1, type: 'square', vol: 0.08 })
   463→    }
   464→  }
   465→
   466→  /* gentle "oops" when a monster slips past — friendly, never scary */
   467→  oops() {
   468→    this.init()
   469→    if (!this.ctx) return
   470→    const t = this.ctx.currentTime
   471→    this.tone({ freq: 392, time: t, dur: 0.15, type: 'triangle', vol: 0.18 })
   472→    this.tone({ freq: 311, time: t + 0.16, dur: 0.28, type: 'triangle', vol: 0.18 })
   473→  }
   474→
   475→  /* little fanfare when a level is complete */
   476→  levelUp() {
   477→    this.init()
   478→    if (!this.ctx) return
   479→    const t = this.ctx.currentTime
   480→    const notes = [72, 76, 79, 84] // C5 E5 G5 C6
   481→    notes.forEach((n, i) =>
   482→      this.tone({ freq: this.f(n), time: t + i * 0.1, dur: i === 3 ? 0.45 : 0.18, type: 'triangle', vol: 0.2 })
   483→    )
   484→  }
   485→
   486→  /* BIG happy fanfare for finishing all three levels */
   487→  win() {
   488→    this.init()
   489→    if (!this.ctx) return
   490→    const t = this.ctx.currentTime
   491→    const seq: Array<[number, number, number]> = [
   492→      [72, 0, 0.14],
   493→      [76, 0.12, 0.14],
   494→      [79, 0.24, 0.14],
   495→      [84, 0.36, 0.3],
   496→      [79, 0.6, 0.12],
   497→      [84, 0.72, 0.14],
   498→      [88, 0.9, 0.6],
   499→    ]
   500→    seq.forEach(([n, dt, dur]) =>
   501→      this.tone({ freq: this.f(n), time: t + dt, dur, type: 'triangle', vol: 0.2 })
   502→    )
   503→  }
   504→
   505→  /* soft, encouraging jingle for "Great try!" — no sad trombones here */
   506→  gameOverJingle() {
   507→    this.init()
   508→    if (!this.ctx) return
   509→    const t = this.ctx.currentTime
   510→    this.tone({ freq: this.f(64), time: t, dur: 0.22, type: 'triangle', vol: 0.16 })
   511→    this.tone({ freq: this.f(60), time: t + 0.2, dur: 0.22, type: 'triangle', vol: 0.16 })
   512→    this.tone({ freq: this.f(57), time: t + 0.42, dur: 0.5, type: 'triangle', vol: 0.16 })
   513→  }
   514→
   515→  /* ── MUSIC LOOP 🎵 — a bouncy 8-bar tune that repeats forever ────────
   516→     32 steps of 8th-notes at 120 BPM. Chords: C → Am → F → G.          */
   517→  startMusic(fast = false) {
   518→    this.init()
   519→    if (!this.ctx || this.musicTimer) return
   520→    this.stepDur = fast ? 0.227 : 0.25 // 132 BPM vs 120 BPM
   521→    this.step = 0
   522→    this.nextTime = this.ctx.currentTime + 0.1
   523→    this.musicTimer = setInterval(() => this.scheduleMusic(), 100)
   524→  }
   525→
   526→  stopMusic() {
   527→    if (this.musicTimer) {
   528→      clearInterval(this.musicTimer)
   529→      this.musicTimer = null
   530→    }
   531→  }
   532→
   533→  private scheduleMusic() {
   534→    if (!this.ctx) return
   535→    const STEP = this.stepDur // one 8th-note at the current tempo
   536→    while (this.nextTime < this.ctx.currentTime + 0.4) {
   537→      this.playMusicStep(this.step, this.nextTime)
   538→      this.step = (this.step + 1) % 32
   539→      this.nextTime += STEP
   540→    }
   541→  }
   542→
   543→  private playMusicStep(s: number, t: number) {
   544→    if (!this.musicGain) return
   545→    // melody (midi notes, 0 = rest) — 4 bars of pure pentatonic happiness
   546→    const MEL = [72, 0, 76, 0, 79, 0, 76, 0, 81, 0, 79, 0, 76, 0, 74, 0, 72, 0, 76, 0, 79, 0, 81, 0, 79, 0, 76, 0, 74, 0, 72, 0]
   547→    // bass roots: C  A  F  G (one per bar)
   548→    const BASS = [48, 45, 41, 43]
   549→    if (MEL[s]) this.tone({ freq: this.f(MEL[s]), time: t, dur: 0.22, type: 'triangle', vol: 0.15, dest: this.musicGain })
   550→    if (s % 8 === 0) this.tone({ freq: this.f(BASS[(s / 8) | 0]), time: t, dur: 0.5, type: 'sine', vol: 0.2, dest: this.musicGain })
   551→    if (s % 8 === 4) this.tone({ freq: this.f(BASS[(s / 8) | 0]) + 12, time: t, dur: 0.3, type: 'sine', vol: 0.1, dest: this.musicGain })
   552→    if (s % 2 === 1) this.tone({ freq: 2100, time: t, dur: 0.03, vol: 0.025, dest: this.musicGain }) // tiny bounce tick
   553→  }
   554→}

   565→function roundRectPath(ctx: CanvasRenderingContext2D, x: number, y: number, w: number, h: number, r: number) {
   566→  ctx.beginPath()
   567→  ctx.moveTo(x + r, y)
   568→  ctx.arcTo(x + w, y, x + w, y + h, r)
   569→  ctx.arcTo(x + w, y + h, x, y + h, r)
   570→  ctx.arcTo(x, y + h, x, y, r)
   571→  ctx.arcTo(x, y, x + w, y, r)
   572→  ctx.closePath()

   575→/* A five-pointed (or any-pointed!) star path */
   576→function starPath(
   577→  ctx: CanvasRenderingContext2D,
   578→  x: number,
   579→  y: number,
   580→  outer: number,
   581→  inner: number,
   582→  points: number,
   583→  rot: number
   584→) {
   585→  ctx.beginPath()
   586→  for (let i = 0; i < points * 2; i++) {
   587→    const rr = i % 2 === 0 ? outer : inner
   588→    const a = rot + (i * Math.PI) / points
   589→    const px = x + Math.cos(a) * rr
   590→    const py = y + Math.sin(a) * rr
   591→    if (i === 0) ctx.moveTo(px, py)
   592→    else ctx.lineTo(px, py)
   593→  }
   594→  ctx.closePath()
   595→}

   597→/* 🥰 THE PAL'S FACE — three moods, used by all four characters:
   598→     smile     = happy default (with cute blinks)
   599→     surprised = "uh-oh!" round eyes + little O mouth (after losing a heart)
   600→     cheer     = squeezed-shut happy eyes + big open grin (level complete!)
   601→   dx / dy move the face inside the character (like Rocky's window). */
   602→function palFace(ctx: CanvasRenderingContext2D, r: number, expr: FaceExpr, blink: boolean, dx = 0, dy = 0) {
   603→  ctx.save()
   604→  ctx.translate(dx, dy)
   605→  const ex = r * 0.36 // eye spacing
   606→  const ey = -r * 0.12 // eye height
   607→  const er = r * 0.22 // eye radius
   608→  if (expr === 'cheer') {
   609→    // ^ ^  two happy squeezed-shut eyes
   610→    ctx.strokeStyle = COLORS.face
   611→    ctx.lineWidth = 2.5
   612→    ctx.lineCap = 'round'
   613→    for (const side of [-1, 1]) {
   614→      ctx.beginPath()
   615→      ctx.arc(side * ex, ey, er, Math.PI * 1.1, Math.PI * 1.9)
   616→      ctx.stroke()
   617→    }
   618→  } else {
   619→    // big white eyes — extra wide when surprised!
   620→    const big = expr === 'surprised' ? 1.3 : 1
   621→    ctx.fillStyle = '#ffffff'
   622→    circle(ctx, -ex, ey, er * big)
   623→    circle(ctx, ex, ey, er * big)
   624→    // pupils — tiny and high when surprised, low and calm when smiling
   625→    ctx.fillStyle = COLORS.face
   626→    const pr = expr === 'surprised' ? er * 0.3 : er * 0.45
   627→    const py = expr === 'surprised' ? ey - er * 0.15 : ey + er * 0.25
   628→    circle(ctx, -ex, py, pr)
   629→    circle(ctx, ex, py, pr)
   630→  }
   631→  // the mouth
   632→  if (expr === 'surprised') {
   633→    // 😮 a little round "O" mouth
   634→    ctx.fillStyle = COLORS.face
   635→    circle(ctx, 0, r * 0.24, r * 0.16)
   636→  } else if (expr === 'cheer') {
   637→    // 🥳 a big open grin
   638→    ctx.fillStyle = COLORS.face
   639→    ctx.beginPath()
   640→    ctx.arc(0, r * 0.06, r * 0.3, Math.PI * 0.12, Math.PI * 0.88)
   641→    ctx.closePath()
   642→    ctx.fill()
   643→  } else {
   644→    // 🙂 the classic smile
   645→    ctx.strokeStyle = COLORS.face
   646→    ctx.lineWidth = 2.5
   647→    ctx.lineCap = 'round'
   648→    ctx.beginPath()
   649→    ctx.arc(0, r * 0.12, r * 0.3, Math.PI * 0.2, Math.PI * 0.8)
   650→    ctx.stroke()
   651→  }
   652→  // rosy cheeks — extra rosy when cheering!
   653→  ctx.fillStyle = expr === 'cheer' ? 'rgba(255,110,150,0.5)' : 'rgba(255,110,150,0.35)'
   654→  circle(ctx, -r * 0.6, r * 0.1, r * 0.14)
   655→  circle(ctx, r * 0.6, r * 0.1, r * 0.14)
   656→  ctx.restore()
   657→}

   976→   kind 0 = round blob with antennae · kind 1 = balloon · kind 2 = star critter */
   977→function drawMonster(ctx: CanvasRenderingContext2D, m: Monster, time: number) {
   978→  const r = m.r
   979→  ctx.save()

   984→  if (m.kind === 0) {
   985→    /* round blob */
   986→    // antennae with golden balls
   987→    ctx.strokeStyle = outline
   988→    ctx.lineWidth = 3
   989→    ctx.lineCap = 'round'
   990→    for (const side of [-1, 1]) {
   991→      ctx.beginPath()
   992→      ctx.moveTo(side * r * 0.4, -r * 0.75)
   993→      ctx.lineTo(side * r * 0.55, -r * 1.25)
   994→      ctx.stroke()

   996→      circle(ctx, side * r * 0.58, -r * 1.32, r * 0.16)
   997→    }
   998→    // body
   999→    ctx.beginPath()
  1000→    ctx.arc(0, 0, r, 0, Math.PI * 2)
  1001→    ctx.fillStyle = m.color
  1002→    ctx.fill()
  1003→    ctx.strokeStyle = outline
  1004→    ctx.lineWidth = 3
  1005→    ctx.stroke()
  1006→    // shine
  1007→    ctx.fillStyle = 'rgba(255,255,255,0.35)'
  1008→    circle(ctx, -r * 0.35, -r * 0.42, r * 0.16)
  1009→    palFace(ctx, r, 'smile', m.blink < 0 && m.blink > -0.15)
  1010→  } else if (m.kind === 1) {
  1011→    /* balloon on a string */
  1012→    ctx.strokeStyle = 'rgba(0,0,0,0.25)'
  1013→    ctx.lineWidth = 2
  1014→    ctx.beginPath()
  1015→    ctx.moveTo(0, r)
  1016→    ctx.quadraticCurveTo(Math.sin(time * 3 + m.wob) * 8, r * 1.5, 0, r * 1.9)
  1017→    ctx.stroke()
  1018→    ctx.beginPath()
  1019→    ctx.ellipse(0, 0, r * 0.82, r, 0, 0, Math.PI * 2)
  1020→    ctx.fillStyle = m.color
  1021→    ctx.fill()
  1022→    ctx.strokeStyle = outline
  1023→    ctx.lineWidth = 3
  1024→    ctx.stroke()

  1026→    ctx.fillStyle = m.color
  1027→    ctx.beginPath()
  1028→    ctx.moveTo(-5, r)
  1029→    ctx.lineTo(5, r)
  1030→    ctx.lineTo(0, r - 8)
  1031→    ctx.closePath()
  1032→    ctx.fill()
  1033→    ctx.fillStyle = 'rgba(255,255,255,0.35)'
  1034→    circle(ctx, -r * 0.3, -r * 0.4, r * 0.14)

  1039→    ctx.fillStyle = m.color
  1040→    ctx.fill()
  1041→    ctx.strokeStyle = outline
  1042→    ctx.lineWidth = 3

  1155→function updateParticles(g: GameData, dt: number) {
  1156→  for (let i = g.parts.length - 1; i >= 0; i--) {
  1157→    const p = g.parts[i]
  1158→    p.life -= dt
  1159→    if (p.life <= 0) {
  1160→      g.parts.splice(i, 1)
  1161→      continue
  1162→    }
  1163→    p.x += p.vx * dt
  1164→    p.y += p.vy * dt
  1165→    p.rot += p.vr * dt
  1166→    if (p.kind === 'confetti') {
  1167→      p.vy += 420 * dt // confetti tumbles down
  1168→      p.vx *= 1 - 1.2 * dt
  1169→    } else if (p.kind === 'sparkle') {
  1170→      p.vy += 140 * dt
  1171→    } else if (p.kind === 'ring') {
  1172→      p.size += 190 * dt // ring grows outward
  1173→    }
  1174→  }
  1175→}

  1177→function drawParticles(ctx: CanvasRenderingContext2D, g: GameData) {
  1178→  for (const p of g.parts) {
  1179→    const a = clamp(p.life / p.maxLife, 0, 1)
  1180→    ctx.globalAlpha = a
  1181→    if (p.kind === 'confetti') {
  1182→      ctx.save()
  1183→      ctx.translate(p.x, p.y)
  1184→      ctx.rotate(p.rot)
  1185→      ctx.fillStyle = p.color
  1186→      ctx.fillRect(-p.size / 2, -p.size / 2, p.size, p.size * 0.62)
  1187→      ctx.restore()
  1188→    } else if (p.kind === 'sparkle') {
  1189→      ctx.fillStyle = p.color
  1190→      starPath(ctx, p.x, p.y, p.size + 2, (p.size + 2) * 0.4, 4, p.rot)
  1191→      ctx.fill()
  1192→    } else if (p.kind === 'ring') {
  1193→      ctx.strokeStyle = p.color
  1194→      ctx.lineWidth = 3.5 * a
  1195→      ctx.beginPath()
  1196→      ctx.arc(p.x, p.y, p.size, 0, Math.PI * 2)
  1197→      ctx.stroke()
  1198→    } else if (p.kind === 'text' && p.text) {
  1199→      ctx.font = `bold 22px ${FONT}`
  1200→      ctx.textAlign = 'center'
  1201→      ctx.textBaseline = 'middle'
  1202→      ctx.lineWidth = 4
  1203→      ctx.strokeStyle = 'rgba(255,255,255,0.9)'
  1204→      ctx.strokeText(p.text, p.x, p.y)
  1205→      ctx.fillStyle = p.color
  1206→      ctx.fillText(p.text, p.x, p.y)
  1207→    }
  1208→    ctx.globalAlpha = 1
  1209→  }
  1210→}
